const residentGradeList = document.getElementById('residentGradeList'); const resident2GradeList = document.getElementById('resident2GradeList'); const arbitratorGradeList = document.getElementById('arbitratorGradeList'); // Clear current options gradeList.innerHTML = ''; // Create a Set of grade impressions to quickly check if a grade should be shown const gradeImpressions = new Set(grades.map(g => g.impression)); // Track which grades we've already added to prevent duplicates const addedGrades = new Set(); // Filter allOptions to only include grades that are in the disease-specific list, avoiding duplicates allOptions.forEach(li => { const gradeValue = li.querySelector('input').value; // Only add if it's in the disease-specific grades and we haven't added it yet if (gradeImpressions.has(gradeValue) && !addedGrades.has(gradeValue)) { // Preserve checked state if this grade was previously checked const newLi = li.cloneNode(true); if (checkedValues.includes(gradeValue)) { newLi.querySelector('input').checked = true; } gradeList.appendChild(newLi); // Mark this grade as added to prevent duplicates addedGrades.add(gradeValue); } }); } else if (selectedCount === 1) { const checkedBox = gradeList.querySelector('.grade-checkbox:checked, .ai-grade-checkbox:checked'); dropdownText.textContent = checkedBox.value; } else { dropdownText.textContent = `${selectedCount} ${defaultLabel.toLowerCase()} selected`; } else if (selectedCount === 1) { const checkedBox = menu.querySelector('.grade-checkbox:checked'); dropdownText.textContent = checkedBox.value; } else { dropdownText.textContent = `${selectedCount} grades selected`; } } if (e.target.classList.contains('ai-model-checkbox')) { // Enforce single selection document.querySelectorAll('.ai-model-checkbox').forEach(cb => { if (cb !== e.target) cb.checked = false; }); const dropdownId = menu.id.replace('List', 'Text'); } else if (selectedCount === 1) { const checkedBox = menu.querySelector('.ai-model-checkbox:checked'); dropdownText.textContent = checkedBox.nextElementSibling.textContent; } else { dropdownText.textContent = `1 model selected`; } } if (e.target.classList.contains('ai-grade-checkbox')) { const dropdownId = menu.id.replace('List', 'Text'); const selectedCount = menu.querySelectorAll('.ai-grade-checkbox:checked').length; const dropdownText = document.getElementById(dropdownId); if (selectedCount === 0) { dropdownText.textContent = 'All AI Grades'; } else if (selectedCount === 1) { const checkedBox = menu.querySelector('.ai-grade-checkbox:checked'); dropdownText.textContent = checkedBox.value; } else { dropdownText.textContent = `${selectedCount} AI grades selected`; } } }); }); // Disable AI controls unless "Has AI Grade" is Yes function toggleAiControls() { const hasAi = document.getElementById('hasAiGradeSelect').value === 'yes'; const aiModelDropdown = document.getElementById('aiModelDropdown'); const aiGradeDropdown = document.getElementById('aiGradeDropdown'); aiModelDropdown.disabled = !hasAi; aiGradeDropdown.disabled = !hasAi; document.querySelectorAll('.ai-model-checkbox').forEach(cb => { cb.disabled = !hasAi; if (!hasAi) cb.checked = false; }); document.querySelectorAll('.ai-grade-checkbox').forEach(cb => { cb.disabled = !hasAi; if (!hasAi) cb.checked = false; }); if (!hasAi) { document.getElementById('aiModelText').textContent = 'All Models'; document.getElementById('aiGradeText').textContent = 'All AI Grades'; } } document.getElementById('hasAiGradeSelect').addEventListener('change', toggleAiControls);
